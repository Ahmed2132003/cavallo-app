47: /// Resolves a notification's [deepLinkType] + [targetId] into a concrete
48: /// route path (for example `business_profile` + `5` -> `/business/5`).
49: ///
50: /// Never throws and never returns null or a malformed path. Returns
51: /// [deepLinkFallbackRoute] (`/home`) when:
52: /// - [deepLinkType] is unknown or blank (a blank type means "navigates
53: ///   nowhere" in the backend contract), or
54: /// - [targetId] is null or not a positive id (every supported type
55: ///   requires one, so a null id would otherwise produce `/business/null`).
56: String resolveDeepLink(String deepLinkType, int? targetId) {
57:   final String? pathPattern = switch (deepLinkType) {
58:     DeepLinkTypes.businessProfile => RouteNames.businessProfilePath,
59:     DeepLinkTypes.postDetail => RouteNames.postDetailPath,
60:     DeepLinkTypes.reelDetail => RouteNames.reelDetailPath,
61:     DeepLinkTypes.productDetail => RouteNames.productDetailPath,
62:     DeepLinkTypes.chatThread => RouteNames.chatThreadPath,
63:     _ => null,
64:   };
65: 
66:   if (pathPattern == null) {
67:     return deepLinkFallbackRoute;
68:   }
69:   if (targetId == null || targetId <= 0) {
70:     return deepLinkFallbackRoute;
71:   }
72: 
73:   return pathPattern.replaceFirst(':${RouteNames.idParam}', '$targetId');
74: }
